def find_bam(sample):
    pattern = re.compile(r"^" + re.escape(sample) + r"(?:[._-]|$)")
    matches = [
        p for p in bams
        if any(pattern.match(part) for part in p.parts)
    ]

        return None

        n = p.name.lower()
        if "genome" in n:
            return 0
        if "primary" in n:
            return 1
        if "candidate" in n:
            return 2
        return 3
    best = min(priority(p) for p in matches)
    choices = [p for p in matches if priority(p) == best]
    if len(choices) != 1:
        raise RuntimeError(
            f"Ambiguous BAM for {sample}:\n" +

    return choices[0]

def find_bam(sample):
    pattern = re.compile(r"^" + re.escape(sample) + r"(?:[._-]|$)")
    matches = [
        p for p in bams
        if any(pattern.match(part) for part in p.parts)
    ]
    if not matches:
        return None
    def priority(p):
        n = p.name.lower()
        if "genome" in n:
            return 0
        if "primary" in n:
            return 1
        if "candidate" in n:
            return 2
        return 3
    best = min(priority(p) for p in matches)
    choices = [p for p in matches if priority(p) == best]
    if len(choices) != 1:
        raise RuntimeError(
            f"Ambiguous BAM for {sample}:\n" +
            "\n".join(map(str, choices))
        )
    return choices[0]

def overlap(blocks, intervals):
    return sum(
        max(0, min(b, d) - max(a, c) + 1)
        for a, b in blocks for c, d in intervals
    )

summary, details, sources = [], [], []

for sample in samples:
    bam = find_bam(sample)
    row = {"Sample": sample, "LCORL_primary_reads": "MISSING"}
    for t in transcripts:
        row[t["name"]] = "MISSING"
    if bam is None:
        summary.append(row)
        sources.append({"Sample": sample, "BAM": "MISSING"})
        continue

    sources.append({"Sample": sample, "BAM": str(bam)})
    command = ["samtools", "view", "-F", "2308", str(bam)]
    # 2308 = unmapped + secondary + supplementary.
    indexed = (
        Path(str(bam) + ".bai").exists()
        or bam.with_suffix(".bai").exists()
        or Path(str(bam) + ".csi").exists()
    )
    if indexed:
        command.append(region)

    result = subprocess.run(command, text=True, capture_output=True)
    if result.returncode:
        raise RuntimeError(result.stderr)

    reads = {}
    for line in result.stdout.splitlines():
        f = line.split("\t")
        if f[2] != "6":
            continue
        pos = int(f[3])
        start = pos
        blocks = []
        junctions = 0
        for length, op in re.findall(r"(\d+)([MIDNSHP=X])", f[5]):
            length = int(length)
            if op in ("M", "=", "X"):
                blocks.append((pos, pos + length - 1))
                pos += length
            elif op in ("D", "N"):
                pos += length
                junctions += (op == "N")
        if start > gene_end or pos - 1 < gene_start:
            continue
        if f[0] in reads:
            raise RuntimeError(f"Duplicate primary read: {sample}, {f[0]}")
        reads[f[0]] = (blocks, int(f[4]), junctions)

    row["LCORL_primary_reads"] = len(reads)
    for t in transcripts:
        cds_len = sum(b-a+1 for a, b in t["cds"])
        covering = 0
        maximum = 0.0
        for read_id, (blocks, mapq, junctions) in reads.items():
            covered = overlap(blocks, t["cds"])
            exon_covered = overlap(blocks, t["exons"])
            percent = 100 * covered / cds_len
            covering += (covered > 0)
            maximum = max(maximum, percent)
            details.append({
                "Sample": sample,
                "Read_ID": read_id,
                "Transcript": t["name"],
                "MAPQ": mapq,
                "Splice_N_count": junctions,
                "Exon_overlap_nt": exon_covered,
                "CDS_overlap_nt": covered,
                "Reference_CDS_nt": cds_len,
                "CDS_coverage_percent": round(percent, 2),
            })
        row[t["name"]] = f"{covering}; {maximum:.1f}%"
    summary.append(row)
